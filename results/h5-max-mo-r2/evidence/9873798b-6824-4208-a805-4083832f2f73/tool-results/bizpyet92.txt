    17	// Runs fn({ call, file, restart, importInto }) with the "trip" group created on a fresh data file.
    18	export async function withTrip(fn) {
    19	  const data = tempDataFile()
    20	  let app = await startApp(data.file)
    21	  try {
    22	    assert.equal((await app.call('POST', '/groups', trip)).status, 201)
    23	    const call = (...args) => app.call(...args)
    24	    const restart = async () => {
    25	      await app.stop()
    26	      app = await startApp(data.file)
    27	    }
    28	    const importInto = (body, group = 'trip') => call('POST', `/groups/${group}/import`, body)
    29	    await fn({ call, file: data.file, restart, importInto })
    30	  } finally {
    31	    await app.stop()
    32	    data.cleanup()
    33	  }
    34	}

    36	// What identifies one write of the data file: a save renames a new file over it, so the inode changes.
    37	export function snapshot(file) {
    38	  const { ino, mtimeMs } = statSync(file)
    39	  return { ino, mtimeMs, text: readFileSync(file, 'utf8') }
    40	}

    42	// Everything the API shows about a group's ledger.
    43	export async function ledger(call, group = 'trip') {
    44	  const [expenses, payments, balances] = await Promise.all(
    45	    ['expenses', 'payments', 'balances'].map(part => call('GET', `/groups/${group}/${part}`)),
    46	  )
    47	  return { ...expenses.body, ...payments.body, ...balances.body }
    48	}

    50	// Asserts that a request left the data file and the ledger exactly as they were.
    51	export async function assertNothingSaved({ call, file }, before) {
    52	  assert.deepEqual(snapshot(file), before.file)
    53	  assert.deepEqual(await ledger(call), before.ledger)
    54	}

   173	// Imports entries (an array, or the raw JSON text of a whole body) and asserts the exact 422 reply, and that
   174	// nothing was saved.
   175	async function assertProblems(ctx, entries, problems, flags = {}) {
   176	  const before = await capture(ctx)
   177	  const r = await ctx.importInto(typeof entries === 'string' ? entries : { entries, ...flags })
   178	  assert.equal(r.status, 422, typeof entries === 'string' ? entries : JSON.stringify(entries))
   179	  assert.deepEqual(r.body, { error: 'import rejected', problems })
   180	  await assertNothingSaved(ctx, before)
   181	}
